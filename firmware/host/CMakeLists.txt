cmake_minimum_required(VERSION 3.16)
project(pocketwayfarer_host C)
set(CMAKE_C_STANDARD 11)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/../common/lv_conf.h" CACHE PATH "" FORCE)
add_subdirectory(../.deps/lvgl lvgl)

set(COMMON ../common/rng.c ../common/timing.c)
file(GLOB COMMON_UI ../common/ui/*.c ../common/ui/fonts/*.c)
set(COMMON_INC ../common ../common/ui)

# ---- Companion game (v0.1) ----
file(GLOB PET_CORE ../pet/core/*.c)
file(GLOB PET_UI ../pet/ui/*.c ../pet/ui/art/*.c)
add_executable(test_pet ../pet/host/test_core.c ${PET_CORE} ${COMMON})
target_include_directories(test_pet PRIVATE ../pet/core ${COMMON_INC})
target_compile_options(test_pet PRIVATE -Wall -Wextra -Werror)
add_executable(preview_pet ../pet/host/preview.c ${PET_CORE} ${PET_UI} ${COMMON} ${COMMON_UI})
target_include_directories(preview_pet PRIVATE ../pet/core ../pet/ui ${COMMON_INC})
target_compile_options(preview_pet PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_link_libraries(preview_pet lvgl m)

# ---- Fantasy RPG ----
file(GLOB RPG_CORE ../rpg/core/*.c)
file(GLOB RPG_UI ../rpg/ui/*.c ../rpg/ui/art/*.c)
if(RPG_CORE)
  add_executable(test_rpg ../rpg/host/test_rpg.c ${RPG_CORE} ${COMMON})
  target_include_directories(test_rpg PRIVATE ../rpg/core ${COMMON_INC})
  target_compile_options(test_rpg PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(test_rpg m)
endif()
if(RPG_UI)
  add_executable(preview_rpg ../rpg/host/preview.c ${RPG_CORE} ${RPG_UI} ${COMMON} ${COMMON_UI})
  target_include_directories(preview_rpg PRIVATE ../rpg/core ../rpg/ui ${COMMON_INC})
  target_compile_options(preview_rpg PRIVATE -Wall -Wextra -Wno-unused-parameter)
  target_link_libraries(preview_rpg lvgl m)
endif()
