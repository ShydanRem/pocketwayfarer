cmake_minimum_required(VERSION 3.16)
project(pocketwayfarer_host C)
set(CMAKE_C_STANDARD 11)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/../common/lv_conf.h" CACHE PATH "" FORCE)
add_subdirectory(../.deps/lvgl lvgl)

set(COMMON ../common/rng.c ../common/timing.c ../common/device_input.c)
file(GLOB COMMON_UI ../common/ui/*.c ../common/ui/fonts/*.c)
set(COMMON_INC ../common ../common/ui)
add_executable(test_device_input test_device_input.c ../common/device_input.c)
target_include_directories(test_device_input PRIVATE ../common)
target_compile_options(test_device_input PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_executable(test_hero_cache test_hero_cache.c ../rpg/ui/hero_cache.c ${COMMON_UI} ${COMMON})
target_include_directories(test_hero_cache PRIVATE ../rpg/ui ${COMMON_INC})
target_compile_options(test_hero_cache PRIVATE -Wall -Wextra -Werror -UNDEBUG -Wno-unused-parameter)
target_link_libraries(test_hero_cache lvgl m)

  add_executable(test_hero_rig ../rpg/host/test_hero_rig.c ../rpg/ui/hero_rig.c ../rpg/ui/art/hero_rig_art.c ../rpg/ui/art/rpg_art.c)

  # firmware and repeatable headless checks do not depend on SDL.
  set(SDL2_HEADER_DIR "${CMAKE_CURRENT_LIST_DIR}/../../build/desktop-deps/include")
  find_library(SDL2_RUNTIME NAMES SDL2 SDL2-2.0)
  if(NOT SDL2_RUNTIME)
    find_file(SDL2_RUNTIME NAMES libSDL2-2.0.so.0 PATHS /lib/x86_64-linux-gnu /usr/lib/x86_64-linux-gnu)
  endif()
  if(EXISTS "${SDL2_HEADER_DIR}/SDL.h" AND SDL2_RUNTIME)
    add_executable(play_lantern ../rpg/host/desktop.c ${RPG_CORE} ${RPG_UI} ${COMMON} ${COMMON_UI})
    target_include_directories(play_lantern PRIVATE ../rpg/core ../rpg/ui ${COMMON_INC} "${SDL2_HEADER_DIR}")
